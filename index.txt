let timer = null;
let confirmedBalance = 0;
let unconfirmedBalance = 0;




async function rpc(category, method, args = []) {
    try {
        const body = args.length ? { args } : undefined;
        const res = await fetch(`/${category}/${method}`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: body ? JSON.stringify(body) : undefined
        });
        const data = await res.json();
        console.log(category, method, data);

        // Balance
        if (method === 'GetBalance') {
            const sats = data.result?.confirmedSats ? parseInt(data.result.confirmedSats, 10) : 0;
            confirmedBalance = sats / 1e8;
        }
        document.getElementById('balance').innerHTML = `${confirmedBalance} BTC`;

        // Result handlers
        if (method === 'GetChainInfo')    showInfo(data.result);
        if (method === 'GetSidechains')   showSidechains(data.result);
        if (method === 'ListTransactions') showInfo(data.result);
        if (method === 'CreateNewAddress') { showAddressQRCode(data.result.address); console.log(data.result.address); }

        return data;
    } catch (err) {
        console.error(`RPC Error: ${category}/${method}`, err);
        return { error: err.toString() };
    }
}


function showInfo(info) {
    let resultDiv = document.getElementById("result");
resultDiv.innerHTML = '';
    // Hide other UI elements
    hide([
        'mnemonicContainer', 'create_wallet', 'walletName', 'confirmWalletCreation',
        'btcAddress', 'feerate', 'scanButton', 'btc', 'amount', 'confirm',
        'qrcode-wrapper', 'qrcode', 'address'
    ]);

    

    // Create container
    const container = document.createElement('div');
    //container.style.background = '#1e1e1e';
    container.style.padding = '15px';
    container.style.borderRadius = '8px';
    container.style.marginTop = '15px';
    container.style.fontFamily = 'Consolas, monospace';
    container.style.fontSize = '34px';
    container.style.color = '#d4d4d4';
    container.style.maxHeight = '60vh';
    container.style.overflowY = 'auto';

    // Check if we have transactions
    if (!info.transactions || !Array.isArray(info.transactions) || info.transactions.length === 0) {
        container.innerHTML = '<p style="color:#ff6b6b;">No transactions found.</p>';
        resultDiv.appendChild(container);
        resultDiv.style.visibility = 'visible';
        resultDiv.style.marginTop = '-1250px';
        return;
    }

    // Sort by confirmation height descending (highest = most recent)
    const sortedTxs = [...info.transactions].sort((a, b) => {
        const heightA = a.confirmationInfo?.height || 0;
        const heightB = b.confirmationInfo?.height || 0;
        return heightB - heightA;
    });

    // Take only the most recent 5
    const recentTxs = sortedTxs.slice(0, 5);

    // Build simple list
    const ul = document.createElement('ul');
    ul.style.listStyle = 'none';
    ul.style.padding = '0';
    ul.style.margin = '0';

    recentTxs.forEach(tx => {
        const li = document.createElement('li');
        li.style.padding = '12px 0';
        li.style.borderBottom = '1px solid #333';

        // Short txid (first 6 chars)
        const shortTxid = tx.txid?.hex?.substring(0, 6) || '??????';

        // Amount: prefer sentSats if present (outgoing), else receivedSats (incoming)
        let amountDisplay = '';
        if (tx.sentSats) {
            const sats = parseInt(tx.sentSats, 10);
            amountDisplay = `<span style="color:#ff6b6b; font-weight:bold;">-${(sats / 1e8).toFixed(8)} BTC</span>`;
        } else if (tx.receivedSats) {
            const sats = parseInt(tx.receivedSats, 10);
            amountDisplay = `<span style="color:#4ec9b0; font-weight:bold;">+${(sats / 1e8).toFixed(8)} BTC</span>`;
        } else {
            amountDisplay = '<span style="color:#888;">?</span>';
        }


        const time = tx.confirmationInfo?.timestamp || 'unconfirmed';

        li.innerHTML = `
            <div style="display:flex; align-items:center; gap:300px;">
                <div>
                    <span style="color:#9cdcfe; font-weight:bold;">${shortTxid}...</span>
              
                </div>
                <div style="text-align:right;">
                    ${amountDisplay}
                </div>
            </div>
            <div style="color:#888; font-size:0.85em; margin-top:4px;">
                ${time}
            </div>
        `;

        ul.appendChild(li);
    });

    container.appendChild(ul);

    // If fewer than 5
    if (recentTxs.length < info.transactions.length) {
        const note = document.createElement('p');
        note.style.color = '#888';
        note.style.fontSize = '0.9em';
        note.style.marginTop = '15px';
       // note.textContent = `Showing most recent ${recentTxs.length} of ${info.transactions.length} total transactions`;
        container.appendChild(note);
    }

    resultDiv.appendChild(container);

    resultDiv.style.visibility = 'visible';
    resultDiv.style.marginTop = '-1250px';
    resultDiv.style.fontSize = '16px'; // readable size
}

// Validate Bitcoin, Testnet, or Signet addresses
function isValidBitcoinAddress(address) {
    // Matches: bc1/tb1/sb1 (segwit) or 1/3 legacy addresses
    return /^(bc1|tb1|sb1|[13])[a-zA-HJ-NP-Z0-9]{20,100}$/.test(address);
}

function handleScannedText(text) {

    // Remove prefixes like bitcoin:, testnet:, signet:
    let clean = text.replace(/^[a-zA-Z]+:/, "");

    // Split off any query params
    let parts = clean.split("?");
    let address = parts[0];

    // Parse amount if included
    let amount = parts[1] ? new URLSearchParams(parts[1]).get("amount") : null;

    // Validate the address
    if (isValidBitcoinAddress(address)) {
        document.getElementById("btcAddress").value = address;
        if (amount) document.getElementById("amount").value = amount;
    } else {
        alert("Invalid Bitcoin, Testnet, or Signet address");
    }
}

    console.log(singleQuotedJson);

    try {
        // Send proper JSON (without single quotes) to server
        const res = await fetch("/wallet/SendTransaction", {
            method: "POST",
            headers: { "Content-Type": "application/json" },

        });

        const data = await res.json();

        if (data.error) {

            resultDiv.innerHTML += `<p style="color:red;">Transaction failed: ${data.error}</p>`;

            resultDiv.innerHTML += `<p style="color:green;">Transaction sent! TXID: ${data.result.txid.hex}</p>`;

        }

function getCurrentSidechain() {

async function detectRunningSidechainsViaProxy() {
    const sidechains = [
        { name: "BitNames", proxy: "bitnames" },
        { name: "BitAssets", proxy: "bitassets" },
        { name: "Thunder", proxy: "thunder" },
        { name: "Truthcoin-DC", proxy: "truthcoin-dc" },
        { name: "ZSide", proxy: "zside" },
        { name: "Photon", proxy: "photon" },
        { name: "Coinshift", proxy: "coinshift" }
    ];

    const running = [];

    for (const sc of sidechains) {
        try {
            const res = await fetch(`/${sc.proxy}/ping`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ jsonrpc: "2.0", id: 1, method: "ping" })
            });

            // Only skip if 404
            if (res.status !== 404 && res.status !== 502) {
                running.push(sc.name);
            } else {
                console.log(`${sc.name} backend returned 404, skipping`);
            }

        } catch (err) {
            console.log(`${sc.name} backend not reachable`, err.message);
        }
    }

    return running; // array of names that are actually reachable
}

    // Add running sidechains except current
    running.forEach(name => {
        const lower = name.toLowerCase();
        if (lower !== current) {
            html += `<li class="sidechain-item" onclick="window.location.href='${lower}.html'">${name}</li>`;
        }
    });

    html += '</ul>';
    resultDiv.innerHTML = html;

    // Style the list nicely
    const items = document.querySelectorAll('.sidechain-item');
    items.forEach(item => {
        item.style.fontSize = '48px';
        item.style.padding = '20px';
        item.style.margin = '10px auto';
        item.style.cursor = 'pointer';
        item.style.background = '#111';
        item.style.fontFamily = 'Consolas, monospace';
        item.style.color = 'white';
        item.style.borderRadius = '12px';
        item.style.transition = 'all 0.2s ease-in-out';
        item.onmouseover = () => { item.style.background = '#333'; item.style.transform = 'scale(1.05)'; };
        item.onmouseout = () => { item.style.background = '#111'; item.style.transform = 'scale(1)'; };
    });

    resultDiv.style.visibility = 'visible';
}

    timer = setInterval(() => {

    }, 5000);
}
